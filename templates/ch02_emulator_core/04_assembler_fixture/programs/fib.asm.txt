; fib.asm — LAB-8 program: computes the Fibonacci sequence in r0 forever
; (bounded by the runner's --cycles budget). Demonstrates a loop that uses
; RAM at 0xE0 as scratch to swap the pair (r0, r1) without a third register.
;
; addr   bytes        mnemonic
; 0x00   10 00 00     LOAD r0, #0x00      ; fib(n)
; 0x03   10 01 01     LOAD r1, #0x01      ; fib(n+1)
; 0x06   20 00 E0     STORE r0, 0xE0      ; <- loop: save old fib(n)
; 0x09   40 00 01     ADD r0, r1          ; r0 = fib(n) + fib(n+1) (wraps)
; 0x0C   30 01 E0     LOADM r1, 0xE0      ; r1 = old fib(n)
; 0x0F   60 06        JMP 0x06            ; next iteration
;
; Never halts by itself — run with --cycles N. Reference golden trace:
; tests/public/ch02_emulator_core/traces/fib.log (--cycles 400).
