cmake_minimum_required(VERSION 3.21)
project(nes_gate_runner CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The reference runner composes VERIFIED solution components from the
# main course templates. LABS_SOLUTIONS_ROOT must point at the generated
# solutions tree (build-solutions/solutions by convention).
if(NOT DEFINED LABS_SOLUTIONS_ROOT)
  # The generator writes the solution tree at the repo root.
  set(LABS_SOLUTIONS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../solutions")
endif()

add_executable(nes_gate_runner nes_gate_runner.cpp)
target_include_directories(nes_gate_runner PRIVATE
  "${LABS_SOLUTIONS_ROOT}"
)
target_compile_options(nes_gate_runner PRIVATE
  $<$<CXX_COMPILER_ID:MSVC>:/W4>
  $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra -Wpedantic>
)

# The gate homebrew is generated by a checked-in Python script (byte-exact,
# deterministic) so the ROM bytes are reproducible from source.
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/gate_homebrew.nes"
  COMMAND ${CMAKE_COMMAND} -E env
          PYTHONPATH="${CMAKE_CURRENT_SOURCE_DIR}/../.."
          python3 "${CMAKE_CURRENT_SOURCE_DIR}/gen_homebrew.py"
          "${CMAKE_CURRENT_BINARY_DIR}/gate_homebrew.nes"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gen_homebrew.py"
  COMMENT "Generating gate homebrew ROM"
  VERBATIM
)
add_custom_target(nes_gate_rom ALL
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/gate_homebrew.nes"
)
