cmake_minimum_required(VERSION 3.21)
project(ps1_gate_runner CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The reference runner composes VERIFIED PS1 solution components from the
# main course templates. LABS_SOLUTIONS_ROOT must point at the generated
# solutions tree (build-solutions/solutions by convention), mirroring
# tools/labs/nes_gate/CMakeLists.txt.
if(NOT DEFINED LABS_SOLUTIONS_ROOT)
  set(LABS_SOLUTIONS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../solutions")
endif()

add_executable(ps1_gate_runner ps1_gate_runner.cpp)
target_include_directories(ps1_gate_runner PRIVATE
  "${LABS_SOLUTIONS_ROOT}"
)
target_compile_options(ps1_gate_runner PRIVATE
  $<$<CXX_COMPILER_ID:MSVC>:/W4>
  $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra -Wpedantic>
)

# Fixture ROMs are generated by a checked-in Python script (byte-exact,
# deterministic) so bytes are reproducible from source.
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ps1_fixtures.stamp"
  COMMAND ${CMAKE_COMMAND} -E env
          PYTHONPATH="${CMAKE_CURRENT_SOURCE_DIR}/../.."
          python3 "${CMAKE_CURRENT_SOURCE_DIR}/gen_ps1_fixtures.py"
          --out-dir "${CMAKE_CURRENT_SOURCE_DIR}/../../.."
  COMMAND ${CMAKE_COMMAND} -E touch "${CMAKE_CURRENT_BINARY_DIR}/ps1_fixtures.stamp"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gen_ps1_fixtures.py"
  COMMENT "Generating PS1 gate fixture ROMs"
  VERBATIM
)
add_custom_target(ps1_gate_fixtures ALL
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/ps1_fixtures.stamp"
)
